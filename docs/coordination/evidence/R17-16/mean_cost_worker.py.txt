"""Small fresh-process mean workload, not the final all-tier scale harness."""
import ctypes
import hashlib
import importlib
import json
import sys
import time
from pathlib import Path

root, count_text, pattern = sys.argv[1:]
root = Path(root).resolve()
sys.path.insert(0, str(root / "python/deepviz/src"))
from deepviz import normalize

count = int(count_text)
value = {"one": 1.0, "tiny": float.fromhex("0x0.0000000000001p-1022"), "max": sys.float_info.max}[pattern]


class Counters(ctypes.Structure):
    _fields_ = [("cb", ctypes.c_ulong), ("faults", ctypes.c_ulong), *[(k, ctypes.c_size_t) for k in ["peak_ws", "ws", "peak_paged", "paged", "peak_nonpaged", "nonpaged", "pagefile", "peak_pagefile"]]]


def memory():
    counters = Counters()
    counters.cb = ctypes.sizeof(counters)
    kernel = ctypes.WinDLL("kernel32", use_last_error=True)
    kernel.GetCurrentProcess.restype = ctypes.c_void_p
    psapi = ctypes.WinDLL("psapi", use_last_error=True)
    psapi.GetProcessMemoryInfo.argtypes = [ctypes.c_void_p, ctypes.POINTER(Counters), ctypes.c_ulong]
    assert psapi.GetProcessMemoryInfo(kernel.GetCurrentProcess(), ctypes.byref(counters), counters.cb)
    return {"working_set": counters.ws, "inclusive_peak_working_set": counters.peak_ws}


module = Path(importlib.import_module("deepviz.normalize").__file__).resolve()
assert module == root / "python/deepviz/src/deepviz/normalize.py"
before = memory()
consumed = 0


def rows():
    global consumed
    for _ in range(count):
        consumed += 1
        yield {"g": "one", "v": value}


start = time.perf_counter()
result = normalize(rows(), {"x": "g", "y": "v", "aggregation": "mean", "max_points": 10})
elapsed = time.perf_counter() - start
after = memory()
assert consumed == result.input_count == count
assert result.strategy == "mean" and result.warnings == []
assert result.records == [{"g": "one", "v": value}]
print(json.dumps({"language": "python", "root": str(root), "source_sha256": hashlib.sha256(module.read_bytes()).hexdigest(), "pattern": pattern, "input_count": result.input_count, "consumed": consumed, "output_count": len(result.records), "expected_hex": value.hex(), "result_hex": result.records[0]["v"].hex(), "elapsed_seconds": elapsed, "before_bytes": before, "after_bytes": after, "method": "fresh process, inclusive Windows PeakWorkingSetSize after timed normalization, interpreter/imports included; not operation allocations or full native/browser cap"}))
