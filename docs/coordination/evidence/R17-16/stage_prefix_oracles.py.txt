def put(path, data, kind, source):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data)
    assert path.read_bytes() == data
    items.append({"path": str(path), "source": source, "kind": kind, "sha256": hashlib.sha256(data).hexdigest(), "lf_sha256": hashlib.sha256(data.replace(b"\r\n", b"\n")).hexdigest()})
