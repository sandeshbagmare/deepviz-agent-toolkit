    ("role09_recipe", "tests/r13_probes/role09/retest-helper-null.ps1"),
    ("role11", "docs/coordination/reports/R10-11-python-scale-envelope.md"),

manifest = {"input_kind": "immutable Git LF blobs; not dirty sibling worktrees", "objects": objects, "files": [], "no_progress_queries": True}
for role, path in specs:
    argv = ["git", "show", objects[role] + ":" + path]
    done = subprocess.run(argv, cwd=ROOT, capture_output=True, timeout=30)
    if done.returncode:
        raise RuntimeError("required_immutable_input_unavailable:" + path)
    target = OUT / "inputs" / role / path
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(done.stdout)
    manifest["files"].append({"role": role, "source_commit": objects[role], "source_path": path, "saved_path": target.relative_to(ROOT).as_posix(), "bytes": len(done.stdout), "sha256": hashlib.sha256(done.stdout).hexdigest(), "git_show_exit": done.returncode})
(OUT / "immutable-inputs.json").write_text(json.dumps(manifest, indent=2) + chr(10), encoding="utf-8")
print(json.dumps(manifest, indent=2))
