records = {}


def copy(source, relative, normalize=False):
    original = source.read_bytes()
    data = original
    if normalize and original.startswith((b"\xff\xfe", b"\xfe\xff")):
        data = original.decode("utf-16").encode("utf-8")
    destination = OUT / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    if destination.exists():
        assert destination.read_bytes() == data, relative
    else:
        destination.write_bytes(data)
    records[relative] = {"bytes": len(data), "sha256": hashlib.sha256(data).hexdigest(),
                         "raw_sha256": hashlib.sha256(original).hexdigest(), "utf16_to_utf8": data != original}
