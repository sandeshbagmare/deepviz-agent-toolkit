def git(args):
    done = subprocess.run(["git", *args], cwd=MAIN, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=60)

def check(name, ok, evidence):
    result["checks"].append({"name": name, "status": "pass" if ok else "fail", "evidence": evidence})
